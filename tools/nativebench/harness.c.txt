/* Checked execution includes session and result cleanup; admission is separate. */
typedef __UINT64_TYPE__ U64;
typedef __UINT32_TYPE__ U32;
typedef struct Context Context;
typedef struct Program Program;
typedef struct Value Value;
typedef struct { U32 status; U32 code; const Value *value; } ResultSlot;
struct timespec { long seconds; long nanoseconds; };
extern const Program *lo_program_new(void);
extern void lo_program_free(const Program *);
extern Context *lo_context_from_program(const Program *, U64);
extern void lo_context_free(Context *);
extern void lo_context_run(Context *, ResultSlot *);
extern void lo_value_release(const Value *);
extern U64 lo_runtime_live_contexts(void);
extern U64 lo_runtime_live_values(void);
extern int lo_cli_main(int, const char **, const char **);
extern int clock_gettime(int, struct timespec *);
extern long write(int, const void *, __SIZE_TYPE__);
static int number(U64 value) {
 char digits[32]; unsigned end = sizeof(digits), start = end;
 do { digits[--start] = '0' + value % 10; value /= 10; } while (value);
 return write(1, digits + start, end - start) == end - start;
}
static U64 now(void) {
 static U64 previous;
 struct timespec time;
#ifdef __APPLE__
 const int clock = 6;
#else
 const int clock = 1;
#endif
 if (clock_gettime(clock, &time) || time.seconds < 0 ||
     time.nanoseconds < 0 || time.nanoseconds >= 1000000000L) return 0;
 U64 nanoseconds = (U64)time.nanoseconds;
 if ((U64)time.seconds > (~(U64)0 - nanoseconds) / 1000000000ULL) return 0;
 U64 current = (U64)time.seconds * 1000000000ULL + nanoseconds;
 if (current < previous) return 0;
 previous = current;
 return current;
}
static int run_batch(const Program *program, U64 batch_size, U64 admitted_values, U64 *elapsed) {
 U64 start = now();
 if (!start) return 15;
 for (U64 index = 0; index < batch_size; ++index) {
  Context *context = lo_context_from_program(program, 10000000);
  if (!context) return 10;
  ResultSlot result = {0, 0, 0};
  lo_context_run(context, &result);
  if (result.status || !result.value) return 11;
  lo_context_free(context);
  lo_value_release(result.value);
 }
 U64 end = now();
 if (!end || end <= start) return 15;
 *elapsed = end - start;
 if (lo_runtime_live_contexts() || lo_runtime_live_values() != admitted_values) return 12;
 return 0;
}
int main(int argc, const char **argv, const char **environment) {
 if (argc == 2) {
  const char expected[] = "--verify";
  unsigned index = 0;
  while (expected[index] && argv[1][index] == expected[index]) ++index;
  if (expected[index] || argv[1][index]) return 7;
  const char *arguments[] = {argv[0], "run", "--max-steps", "10000000"};
  return lo_cli_main(4, arguments, environment);
 }
 if (argc != 1) return 7;
 const Program *program = lo_program_new();
 if (!program) return 9;
 U64 admitted_values = lo_runtime_live_values();
 U64 batch_size = 1, elapsed;
 for (;;) {
  int status = run_batch(program, batch_size, admitted_values, &elapsed);
  if (status) return status;
  if (elapsed >= 1000000ULL || batch_size == 64) break;
  batch_size *= 2;
 }
 U64 total = 0, iterations = 0;
 do {
  int status = run_batch(program, batch_size, admitted_values, &elapsed);
  if (status) return status;
  total += elapsed;
  iterations += batch_size;
 } while (total < 200000000ULL && iterations < 1000000);
 lo_program_free(program);
 if (lo_runtime_live_values()) return 14;
 const char first[] = "{\"ns\":";
 const char second[] = ",\"iterations\":";
 const char third[] = ",\"batch_size\":";
 const char fourth[] = ",\"total_ns\":";
 if (write(1, first, sizeof(first)-1) != sizeof(first)-1 || !number(total / iterations) ||
     write(1, second, sizeof(second)-1) != sizeof(second)-1 || !number(iterations) ||
     write(1, third, sizeof(third)-1) != sizeof(third)-1 || !number(batch_size) ||
     write(1, fourth, sizeof(fourth)-1) != sizeof(fourth)-1 || !number(total) ||
     write(1, "}\n", 2) != 2) return 13;
 return 0;
}
